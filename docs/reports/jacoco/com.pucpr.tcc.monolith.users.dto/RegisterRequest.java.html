package com.pucpr.tcc.monolith.users.dto;

import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

/** DTO de entrada para cadastro de novo usuário. */
public record RegisterRequest(
    @NotBlank(message = "O nome é obrigatório.") String name,
    @NotBlank(message = "O e-mail é obrigatório.")
    @Email(message = "Formato de e-mail inválido.") String email,
    @NotBlank(message = "A senha é obrigatória.")
    @Size(min = 6, message = "A senha deve ter no mínimo 6 caracteres.") String password
) {}
